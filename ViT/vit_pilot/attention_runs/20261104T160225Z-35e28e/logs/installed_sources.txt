153: def _build_segmented(
154:     gm: torch.fx.GraphModule,
155:     args: tuple[torch.Tensor, ...],
156:     compiler_config,
157:     pass_context: Optional[PassContext],
158: ) -> tuple[torch.fx.GraphModule, Callable]:
159:     """Prepare the split graph and input filter for ``RunnableSequence``.
160: 
161:     DFG tensors are inlined as constants first, so the returned filter removes
162:     them from runtime inputs.
163:     """
164:     ep, filter_inputs = _inline_dfg_tensors_from_user_inputs(gm, args)
165:     inlined_gm = ep.module(check_guards=False)
166: 
167:     split_gm = _split_by_runnable(inlined_gm)
168:     logger.debug(f"building segmented gm, split {split_gm.graph}")
169: 
170:     segment_names = {
171:         name
172:         for name, sub in split_gm.named_children()
173:         if isinstance(sub, torch.fx.GraphModule)
174:         and not any(is_runnable_node(n) for n in sub.graph.nodes)
175:     }
176:     logger.debug(
177:         f"split into {len(list(split_gm.named_children()))} submodules, "
178:         f"{len(segment_names)} to compile"
179:     )
180: 
181:     for name in segment_names:
182:         submod = getattr(split_gm, name)
183:         seg_args = tuple(
184:             _example_tensor(n) for n in submod.graph.find_nodes(op="placeholder")
185:         )
186:         cm = CompileModule.from_module(
187:             submod,
188:             seg_args,
189:             compiler_config=compiler_config,
190:             pass_context=pass_context,
191:         )
192:         # Empty binaries have no device work to move.
193:         if cm.runnable._inner.binary.numel() > 0:
194:             device = _utils.ensure_furiosa_device(seg_args)
195:             cm.to(device)
196:         cm._segment_example_inputs = seg_args  # for `flatten_inputs` order recovery
197:         setattr(split_gm, name, cm)
198: 
199:     return split_gm, filter_inputs

202: def _backend_inner(
203:     gm: torch.fx.GraphModule,
204:     compiler_config=None,
205:     eager_fallback=False,
206:     *,
207:     pass_context: Optional[PassContext] = None,
208: ):
209:     logger.debug(f"torch.compile {gm.graph}")
210:     seq_plan = None
211: 
212:     def forward(*args):
213:         nonlocal gm, compiler_config, seq_plan
214: 
215:         # Build the segmented RunnableSequence once; later calls only bind inputs.
216:         try:
217:             if seq_plan is None:
218:                 split_gm, filter_inputs = _build_segmented(
219:                     gm, args, compiler_config, pass_context
220:                 )
221:                 seq_plan = build_sequence(split_gm, filter_inputs(args))
222:                 seq_plan.filter_inputs = filter_inputs
223:             return run_sequence(seq_plan, args)
224:         except (error.UnsupportedOpError, NotImplementedError) as e:
225:             if not eager_fallback:
226:                 # Implicit eager execution is disabled, re-raise the exception
227:                 logger.error(

229:                 )
230:                 raise e from None
231:             logger.debug(f"eager fallback for the target gm: {e}")
232:             return gm(*args)

253:             nn.init.normal_(
254:                 self.conv_proj.conv_last.weight, mean=0.0, std=math.sqrt(2.0 / self.conv_proj.conv_last.out_channels)
255:             )
256:             if self.conv_proj.conv_last.bias is not None:
257:                 nn.init.zeros_(self.conv_proj.conv_last.bias)
258: 
259:         if hasattr(self.heads, "pre_logits") and isinstance(self.heads.pre_logits, nn.Linear):
260:             fan_in = self.heads.pre_logits.in_features
261:             nn.init.trunc_normal_(self.heads.pre_logits.weight, std=math.sqrt(1 / fan_in))
262:             nn.init.zeros_(self.heads.pre_logits.bias)
263: 
264:         if isinstance(self.heads.head, nn.Linear):
265:             nn.init.zeros_(self.heads.head.weight)
266:             nn.init.zeros_(self.heads.head.bias)
267: 
268:     def _process_input(self, x: torch.Tensor) -> torch.Tensor:
269:         n, c, h, w = x.shape
270:         p = self.patch_size
271:         torch._assert(h == self.image_size, f"Wrong image height! Expected {self.image_size} but got {h}!")
272:         torch._assert(w == self.image_size, f"Wrong image width! Expected {self.image_size} but got {w}!")
273:         n_h = h // p
274:         n_w = w // p
275: 
276:         # (n, c, h, w) -> (n, hidden_dim, n_h, n_w)
277:         x = self.conv_proj(x)
278:         # (n, hidden_dim, n_h, n_w) -> (n, hidden_dim, (n_h * n_w))
279:         x = x.reshape(n, self.hidden_dim, n_h * n_w)
280: 
281:         # (n, hidden_dim, (n_h * n_w)) -> (n, (n_h * n_w), hidden_dim)
282:         # The self attention layer expects inputs in the format (N, S, E)
283:         # where S is the source sequence length, N is the batch size, E is the
284:         # embedding dimension
285:         x = x.permute(0, 2, 1)
286: 
287:         return x
288: 
289:     def forward(self, x: torch.Tensor):
290:         # Reshape and permute the input tensor
291:         x = self._process_input(x)
292:         n = x.shape[0]
293: 
294:         # Expand the class token to the full batch
295:         batch_class_token = self.class_token.expand(n, -1, -1)
296:         x = torch.cat([batch_class_token, x], dim=1)
297: 
298:         x = self.encoder(x)
299: 
300:         # Classifier "token" as used by standard language architectures
301:         x = x[:, 0]
302: 
303:         x = self.heads(x)
304: 
305:         return x

801:         patch_size (int): Patch size of the new model.
802:         model_state (OrderedDict[str, torch.Tensor]): State dict of the pre-trained model.
803:         interpolation_mode (str): The algorithm used for upsampling. Default: bicubic.
804:         reset_heads (bool): If true, not copying the state of heads. Default: False.
805: 
806:     Returns:
807:         OrderedDict[str, torch.Tensor]: A state dict which can be loaded into the new model.
808:     """
809:     # Shape of pos_embedding is (1, seq_length, hidden_dim)
810:     pos_embedding = model_state["encoder.pos_embedding"]
811:     n, seq_length, hidden_dim = pos_embedding.shape
812:     if n != 1:
813:         raise ValueError(f"Unexpected position embedding shape: {pos_embedding.shape}")
814: 
815:     new_seq_length = (image_size // patch_size) ** 2 + 1
816: 
817:     # Need to interpolate the weights for the position embedding.
818:     # We do this by reshaping the positions embeddings to a 2d grid, performing
819:     # an interpolation in the (h, w) space and then reshaping back to a 1d grid.
820:     if new_seq_length != seq_length:
821:         # The class token embedding shouldn't be interpolated, so we split it up.
822:         seq_length -= 1
823:         new_seq_length -= 1
824:         pos_embedding_token = pos_embedding[:, :1, :]
825:         pos_embedding_img = pos_embedding[:, 1:, :]
826: 
827:         # (1, seq_length, hidden_dim) -> (1, hidden_dim, seq_length)
828:         pos_embedding_img = pos_embedding_img.permute(0, 2, 1)
829:         seq_length_1d = int(math.sqrt(seq_length))
830:         if seq_length_1d * seq_length_1d != seq_length:
831:             raise ValueError(
832:                 f"seq_length is not a perfect square! Instead got seq_length_1d * seq_length_1d = {seq_length_1d * seq_length_1d } and seq_length = {seq_length}"
833:             )
834: 
835:         # (1, hidden_dim, seq_length) -> (1, hidden_dim, seq_l_1d, seq_l_1d)
836:         pos_embedding_img = pos_embedding_img.reshape(1, hidden_dim, seq_length_1d, seq_length_1d)
837:         new_seq_length_1d = image_size // patch_size
838: 
839:         # Perform interpolation.
840:         # (1, hidden_dim, seq_l_1d, seq_l_1d) -> (1, hidden_dim, new_seq_l_1d, new_seq_l_1d)
841:         new_pos_embedding_img = nn.functional.interpolate(
842:             pos_embedding_img,
843:             size=new_seq_length_1d,
844:             mode=interpolation_mode,
845:             align_corners=True,
846:         )
847: 
848:         # (1, hidden_dim, new_seq_l_1d, new_seq_l_1d) -> (1, hidden_dim, new_seq_length)
849:         new_pos_embedding_img = new_pos_embedding_img.reshape(1, hidden_dim, new_seq_length)
850: 
851:         # (1, hidden_dim, new_seq_length) -> (1, new_seq_length, hidden_dim)
852:         new_pos_embedding_img = new_pos_embedding_img.permute(0, 2, 1)
853:         new_pos_embedding = torch.cat([pos_embedding_token, new_pos_embedding_img], dim=1)
854: 
855:         model_state["encoder.pos_embedding"] = new_pos_embedding
856: 
857:         if reset_heads:
858:             model_state_copy: "OrderedDict[str, torch.Tensor]" = OrderedDict()
859:             for k, v in model_state.items():
860:                 if not k.startswith("heads"):
861:                     model_state_copy[k] = v
862:             model_state = model_state_copy
863: 
864:     return model_state

 135: class AtenOp:
 136:     def __init__(
 137:         self,
 138:         aten_op_name,
 139:         op_callable,
 140:         args,
 141:         kwargs,
 142:     ):
 143:         self.aten_op_name = aten_op_name
 144:         self.op_callable = op_callable
 145: 
 146:         flattened_args, self.args_spec = pytree.tree_flatten(args)
 147: 
 148:         self.non_traceable_args = {}
 149:         traceable_args = []
 150:         for index, arg in enumerate(flattened_args):
 151:             if isinstance(arg, torch.Tensor):
 152:                 traceable_args.append(arg)
 153:             else:
 154:                 self.non_traceable_args[index] = arg
 155:         self.traceable_args = tuple(traceable_args)
 156: 
 157:         self.non_traceable_kwargs = {}
 158:         self.traceable_kwargs = {}
 159:         for key, value in kwargs.items():
 160:             if isinstance(value, torch.Tensor):
 161:                 self.traceable_kwargs[key] = value
 162:             else:
 163:                 self.non_traceable_kwargs[key] = value
 164: 
 165:         self.args = args
 166:         self.flatten_shape_params = cache_key_from_args(args, kwargs)
 167:         self.aten_op_hash = hash((self.aten_op_name, self.flatten_shape_params))

 170:         return _build_exported_program_cached(self)

 173:         return _compile_and_load_cached(self, device).out()

 243:     """Build an unsharded strided-view handle for ``aten.slice.Tensor``.
 244: 
 245:     Rust composes the resulting byte offset and shape with later views.
 246:     """

 331:             return shapes[handle.out] if 0 <= handle.out < len(shapes) else None

 336:         if isinstance(handle, _Const):

 341:         # Slice/passthrough lowering builds a broadcast (unsharded) DRAM shape,
 342:         # which silently drops inter-chip axes. Reject rather than emit wrong

 344:         base = _source_dram_shape(src)

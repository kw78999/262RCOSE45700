1088: class MultiheadAttention(Module):
1089:     r"""Allows the model to jointly attend to information from different representation subspaces.
1090: 
1091:     This MultiheadAttention layer implements the original architecture described
1092:     in the `Attention Is All You Need <https://arxiv.org/abs/1706.03762>`_ paper. The
1093:     intent of this layer is as a reference implementation for foundational understanding
1094:     and thus it contains only limited features relative to newer architectures.
1095:     Given the fast pace of innovation in transformer-like architectures, we recommend
1096:     exploring this `tutorial <https://pytorch.org/tutorials/intermediate/transformer_building_blocks.html>`_
1097:     to build efficient layers from building blocks in core or using higher
1098:     level libraries from the `PyTorch Ecosystem <https://landscape.pytorch.org/>`_.
1099: 
1100:     Multi-Head Attention is defined as:
1101: 
1102:     .. math::
1103:         \text{MultiHead}(Q, K, V) = \text{Concat}(\text{head}_1,\dots,\text{head}_h)W^O
1104: 
1105:     where :math:`\text{head}_i = \text{Attention}(QW_i^Q, KW_i^K, VW_i^V)`.
1106: 
1107:     ``nn.MultiheadAttention`` will use the optimized implementations of
1108:     ``scaled_dot_product_attention()`` when possible.
1109: 
1110:     In addition to support for the new ``scaled_dot_product_attention()``
1111:     function, for speeding up Inference, MHA will use
1112:     fastpath inference with support for Nested Tensors, iff:
1113: 
1114:     - self attention is being computed (i.e., ``query``, ``key``, and ``value`` are the same tensor).
1115:     - inputs are batched (3D) with ``batch_first==True``
1116:     - Either autograd is disabled (using ``torch.inference_mode`` or ``torch.no_grad``) or no tensor argument ``requires_grad``
1117:     - training is disabled (using ``.eval()``)
1118:     - ``add_bias_kv`` is ``False``
1119:     - ``add_zero_attn`` is ``False``
1120:     - ``kdim`` and ``vdim`` are equal to ``embed_dim``
1121:     - if a `NestedTensor <https://pytorch.org/docs/stable/nested.html>`_ is passed, neither ``key_padding_mask``
1122:       nor ``attn_mask`` is passed
1123:     - autocast is disabled
1124: 
1125:     If the optimized inference fastpath implementation is in use, a
1126:     `NestedTensor <https://pytorch.org/docs/stable/nested.html>`_ can be passed for
1127:     ``query``/``key``/``value`` to represent padding more efficiently than using a
1128:     padding mask. In this case, a `NestedTensor <https://pytorch.org/docs/stable/nested.html>`_
1129:     will be returned, and an additional speedup proportional to the fraction of the input
1130:     that is padding can be expected.
1131: 
1132:     Args:
1133:         embed_dim: Total dimension of the model.
1134:         num_heads: Number of parallel attention heads. Note that ``embed_dim`` will be split
1135:             across ``num_heads`` (i.e. each head will have dimension ``embed_dim // num_heads``).
1136:         dropout: Dropout probability on ``attn_output_weights``. Default: ``0.0`` (no dropout).
1137:         bias: If specified, adds bias to input / output projection layers. Default: ``True``.
1138:         add_bias_kv: If specified, adds bias to the key and value sequences at dim=0. Default: ``False``.
1139:         add_zero_attn: If specified, adds a new batch of zeros to the key and value sequences at dim=1.
1140:             Default: ``False``.
1141:         kdim: Total number of features for keys. Default: ``None`` (uses ``kdim=embed_dim``).
1142:         vdim: Total number of features for values. Default: ``None`` (uses ``vdim=embed_dim``).
1143:         batch_first: If ``True``, then the input and output tensors are provided
1144:             as (batch, seq, feature). Default: ``False`` (seq, batch, feature).
1145: 
1146:     Examples::
1147: 
1148:         >>> # xdoctest: +SKIP
1149:         >>> multihead_attn = nn.MultiheadAttention(embed_dim, num_heads)
1150:         >>> attn_output, attn_output_weights = multihead_attn(query, key, value)
1151: 
1152:     .. _`FlashAttention: Fast and Memory-Efficient Exact Attention with IO-Awareness`:
1153:          https://arxiv.org/abs/2205.14135
1154: 
1155:     """
1156: 
1157:     __constants__ = ["batch_first"]
1158:     bias_k: torch.Tensor | None
1159:     bias_v: torch.Tensor | None
1160: 
1161:     def __init__(
1162:         self,
1163:         embed_dim,
1164:         num_heads,
1165:         dropout=0.0,
1166:         bias=True,
1167:         add_bias_kv=False,
1168:         add_zero_attn=False,
1169:         kdim=None,
1170:         vdim=None,
1171:         batch_first=False,
1172:         device=None,
1173:         dtype=None,
1174:     ) -> None:
1175:         if embed_dim <= 0 or num_heads <= 0:
1176:             raise ValueError(
1177:                 f"embed_dim and num_heads must be greater than 0,"
1178:                 f" got embed_dim={embed_dim} and num_heads={num_heads} instead"
1179:             )
1180:         factory_kwargs = {"device": device, "dtype": dtype}
1181:         super().__init__()
1182:         self.embed_dim = embed_dim
1183:         self.kdim = kdim if kdim is not None else embed_dim
1184:         self.vdim = vdim if vdim is not None else embed_dim
1185:         self._qkv_same_embed_dim = self.kdim == embed_dim and self.vdim == embed_dim

1225:         else:

1517:     def merge_masks(
1518:         self,
1519:         attn_mask: Tensor | None,
1520:         key_padding_mask: Tensor | None,
1521:         query: Tensor,
1522:     ) -> tuple[Tensor | None, int | None]:
1523:         r"""Determine mask type and combine masks if necessary.
1524: 
1525:         If only one mask is provided, that mask
1526:         and the corresponding mask type will be returned. If both masks are provided, they will be both
1527:         expanded to shape ``(batch_size, num_heads, seq_len, seq_len)``, combined with logical ``or``
1528:         and mask type 2 will be returned
1529:         Args:
1530:             attn_mask: attention mask of shape ``(seq_len, seq_len)``, mask type 0
1531:             key_padding_mask: padding mask of shape ``(batch_size, seq_len)``, mask type 1
1532:             query: query embeddings of shape ``(batch_size, seq_len, embed_dim)``
1533:         Returns:
1534:             merged_mask: merged mask
1535:             mask_type: merged mask type (0, 1, or 2)
1536:         """
1537:         mask_type: int | None = None
1538:         merged_mask: Tensor | None = None
1539: 
1540:         if key_padding_mask is not None:
1541:             mask_type = 1
1542:             merged_mask = key_padding_mask
1543: 
1544:         if attn_mask is not None:
1545:             # In this branch query can't be a nested tensor, so it has a shape
1546:             batch_size, seq_len, _ = query.shape
1547:             mask_type = 2
1548: 
1549:             # Always expands attn_mask to 4D
1550:             if attn_mask.dim() == 3:
1551:                 attn_mask_expanded = attn_mask.view(batch_size, -1, seq_len, seq_len)
1552:             else:  # attn_mask.dim() == 2:
1553:                 attn_mask_expanded = attn_mask.view(1, 1, seq_len, seq_len).expand(
1554:                     batch_size, self.num_heads, -1, -1
1555:                 )
1556:             merged_mask = attn_mask_expanded
1557: 
1558:             if key_padding_mask is not None:
1559:                 key_padding_mask_expanded = key_padding_mask.view(
1560:                     batch_size, 1, 1, seq_len
1561:                 ).expand(-1, self.num_heads, -1, -1)
1562:                 merged_mask = attn_mask_expanded + key_padding_mask_expanded
1563: 
1564:         # no attn_mask and no key_padding_mask, returns None, None
1565:         return merged_mask, mask_type
